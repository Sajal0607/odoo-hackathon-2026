@app.route("/attendance") @login_required def attendance(): db = get_db() # Always initialize rows before using it rows = db.execute( """ SELECT date, check_in, check_out, status FROM attendance WHERE employee_id = ? ORDER BY date DESC """, (current_user()["id"],) ).fetchall() today = datetime.now().strftime("%Y-%m-%d") today_att = db.execute( """ SELECT date, check_in, check_out, status FROM attendance WHERE employee_id = ? AND date = ? """, (current_user()["id"], today) ).fetchone() return render_template( "attendance.html", rows=rows, today_att=today_att )
